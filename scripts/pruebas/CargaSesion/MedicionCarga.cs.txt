sealed class Auth(ClaimsPrincipal user):AuthenticationStateProvider{public override Task<AuthenticationState> GetAuthenticationStateAsync()=>Task.FromResult(new AuthenticationState(user));}
sealed class Respuestas:HttpMessageHandler {
 int activas,maximo,cuenta;public int Maximo=>maximo;public int Cuenta=>cuenta;
 protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request,CancellationToken ct){
 Interlocked.Increment(ref cuenta);int actual=Interlocked.Increment(ref activas),old;
 do{old=maximo;}while(actual>old&&Interlocked.CompareExchange(ref maximo,actual,old)!=old);
 try{await Task.Delay(30,ct);return new(HttpStatusCode.OK){Content=new StringContent(request.RequestUri!.AbsolutePath.StartsWith("/PostulanteRS/")?"[{\"id\":111}]":"[]",Encoding.UTF8,"application/json")};}finally{Interlocked.Decrement(ref activas);}}
}
